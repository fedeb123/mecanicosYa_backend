# Respuestas sin token, con otro rol y de un mecánico pendiente

$ curl localhost:8083/api/dispatches/offers   (sin token)
{"type":"about:blank","title":"Unauthorized","status":401,"detail":"Missing, expired or invalid access token"}
HTTP 401

$ curl -H "Authorization: Bearer <mecánico pendiente>" localhost:8083/api/dispatches/offers
{"detail":"Mechanic pending verification: an admin must approve them first","instance":"/api/dispatches/offers","status":403,"title":"Forbidden"}
HTTP 403

$ curl -X POST -H "Authorization: Bearer <mecánico pendiente>" localhost:8082/api/assistances   (rol equivocado)
{"type":"about:blank","title":"Forbidden","status":403,"detail":"Your role cannot perform this operation"}
HTTP 403

$ curl -X POST localhost:8084/api/auth/login -d {"email":"admin@mecanicosya.local","password":"mala"}
{"detail":"Invalid email or password","instance":"/api/auth/login","status":401,"title":"Unauthorized"}
HTTP 401
