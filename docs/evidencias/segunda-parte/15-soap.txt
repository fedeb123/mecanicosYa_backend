# Servicio SOAP de clasificación (assistance-service)
# 2026-11-16T06:06Z. Los cuatro servicios corriendo con Postgres y Kafka del compose; token de un repartidor recién registrado.
# Sin ANTHROPIC_API_KEY clasifica el respaldo por palabras clave (source KEYWORDS); con la clave responde source AI.

## 0. El WSDL se lee sin token
$ curl -s http://localhost:8082/ws/triage.wsdl   (extracto; completo en docs/contratos/triage.wsdl)
      <soap:address location="http://localhost:8082/ws"/>
      <wsdl:input name="ClassifyProblemRequest">
      <wsdl:output name="ClassifyProblemResponse">
    <wsdl:operation name="ClassifyProblem">
  <wsdl:service name="TriageService">

## 1. Sobre válido, con token
$ curl -s -X POST http://localhost:8082/ws -H "Content-Type: text/xml; charset=utf-8" -H "Authorization: Bearer $TOKEN" --data-binary @pedido.xml

## 3. Sobre que no cumple el esquema (vehicleType TRUCK): SOAP Fault de cliente, no llega al endpoint

<SOAP-ENV:Envelope xmlns:SOAP-ENV="http://schemas.xmlsoap.org/soap/envelope/">
  <SOAP-ENV:Header/>
  <SOAP-ENV:Body>
    <SOAP-ENV:Fault>
      <faultcode>SOAP-ENV:Client</faultcode>
      <faultstring xml:lang="en">Validation error</faultstring>
      <detail>
        <spring-ws:ValidationError xmlns:spring-ws="http://springframework.org/spring-ws">cvc-enumeration-valid: Value 'TRUCK' is not facet-valid with respect to enumeration '[BICYCLE, E_BIKE, MOTORCYCLE]'. It must be a value from the enumeration.</spring-ws:ValidationError>
        <spring-ws:ValidationError xmlns:spring-ws="http://springframework.org/spring-ws">cvc-type.3.1.3: The value 'TRUCK' of element 'tri:vehicleType' is not valid.</spring-ws:ValidationError>
      </detail>
    </SOAP-ENV:Fault>
