$ curl -X POST http://localhost:8081/api/mechanics -H 'X-Correlation-Id: ev-errores-1791850376' -H 'Content-Type: application/json' -d '{"name":"","email":"no-es-un-email","latitude":120,"longitude":-58.38,"skills":[]}'
HTTP 400 · 0.04s
{
  "detail": "The request has 4 invalid field(s)",
  "instance": "/api/mechanics",
  "status": 400,
  "title": "Invalid mechanic data",
  "errors": {
    "email": "must be a well-formed email address",
    "latitude": "must be less than or equal to 90.0",
    "name": "must not be blank",
    "skills": "must not be empty"
  }
}
